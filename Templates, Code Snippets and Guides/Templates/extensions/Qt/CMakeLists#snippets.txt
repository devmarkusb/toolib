
# ...

include (qt.cmake)
find_package(Qt5Quick) # Widgets contains a.o.t. Core and Gui
find_package(Qt5LinguistTools)

# ...

set(uiwrap_USE_IMPLEMENTATION       "qt" CACHE STRING "")

# ...

if (uiwrap_USE_IMPLEMENTATION STREQUAL "qt")
    target_include_directories(${TargetAppNameAsLibForTest} SYSTEM PUBLIC ${Qt5Quick_INCLUDE_DIRS})
endif ()

# ...

if (uiwrap_USE_IMPLEMENTATION STREQUAL "qt")
    target_link_libraries(${TargetAppNameAsLibForTest} PUBLIC Qt5::Quick)
endif ()

# ...

# Only purpose of this so far: prevent showing an additional console window in Qt apps (happens for msvc in way
# impossible to overlook, but also for mingw builds when not starting from IDE).
if (WIN32)
    set(WIN32_ON_OFF_SWITCH "WIN32")
else ()
    set(WIN32_ON_OFF_SWITCH "")
endif ()

qt5_add_resources(TargetAppName_RESOURCES gui/Qt/qml.qrc)

include(lang.cmake)

add_executable(${TargetAppName}
# ...
    ${TargetAppName_RESOURCES}
    ${TargetAppName_TRANSLATIONS}
)

add_custom_command(TARGET ${TargetAppName} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory ${OutputDir_bin}/translations)
foreach(translation_file ${TargetAppName_TRANSLATIONS})
    add_custom_command(TARGET ${TargetAppName} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy ${translation_file} ${OutputDir_bin}/translations
    )
endforeach()

# postbuild step before running the program or its test (here we use the program exe to determine needed
# binaries for both the program and the test exe - that was sufficient so far)
if (WIN32)
    add_custom_command(
        TARGET ${TargetAppNameTest} POST_BUILD
        COMMAND qtdeploy_autodepends_win.bat "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}" "$ENV{dev_qt_ver_bin}"
            --qmldir ..\\gui\\Qt $<TARGET_FILE_NAME:${TargetAppName}>
        WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
        )
endif ()
